
FILE arm_cli.py
"""팔 명령 실행기 — 터틀봇 작업 제어기가 필요할 때 이 파일을 실행(또는 import)해서 팔을 움직임. 11/25

명령줄 (작업 제어기에서 subprocess로):
  ~/arm-venv/bin/python ~/so101-mobile-pick/arm_cli.py pick --color yellow
  ~/arm-venv/bin/python ~/so101-mobile-pick/arm_cli.py place
  ~/arm-venv/bin/python ~/so101-mobile-pick/arm_cli.py park
  ~/arm-venv/bin/python ~/so101-mobile-pick/arm_cli.py status

  - 동작이 끝나면 **표준출력에 JSON 한 줄**을 내고 끝남. 진행 로그는 표준에러로.
      {"ok": true, "cmd": "pick", "result": "picked", "fwd_m": 0.21, ...}
      {"ok": false, "cmd": "pick", "result": "not_found", ...}
      {"ok": false, "cmd": "place", "result": "error", "error": "..."}
  - 종료 코드: 0 = 성공(picked/placed/parked), 1 = 실패 결과(not_found/too_far/missed), 2 = 오류
  - pick 결과 picked: 블록을 든 채 운반 자세로 **토크를 켠 채 프로그램만 끝남** → 터틀봇 이동 → 나중에 place 실행.
    들고 있는지·자세는 state/arm_state.json에 남겨 다음 실행이 이어받음.
  - 동시에 두 개를 실행하면 두 번째는 바로 오류(종료 코드 2, "busy").

파이썬에서 직접 (작업 제어기가 같은 venv 파이썬일 때):
  from arm_cli import run
  r = run("pick", color="yellow")     # 위와 같은 dict

pick 결과:
  picked     집어서 운반 자세로 들고 있음 → 목적지로 이동 후 place
  not_found  좌우 60°를 봐도 없음 → 터틀봇을 돌리거나 옮긴 뒤 다시 pick
  too_far    보이지만 안 닿음 → fwd_m/left_m(팔 회전축 기준, 앞·왼쪽 +, m)만큼 다가간 뒤 다시 pick
  missed     못 쥠 (스스로 --retries번 더 해 본 뒤) → 다시 pick
"""
import argparse
import contextlib
import fcntl
import json
import sys
import time
import traceback
from pathlib import Path

ROOT = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT))
STATE_FILE = ROOT / "state/arm_state.json"
LOCK_FILE = ROOT / "state/arm.lock"
EXIT = {"picked": 0, "placed": 0, "parked": 0, "not_found": 1, "too_far": 1, "missed": 1}


def _load_state():
    try:
        return json.loads(STATE_FILE.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError):
        return {}


def _save_state(tasks):
    STATE_FILE.parent.mkdir(parents=True, exist_ok=True)
    tmp = STATE_FILE.with_suffix(".tmp")
    tmp.write_text(json.dumps({"holding": tasks.holding, "pose": tasks.pose, "time": time.time()}), encoding="utf-8")
    tmp.replace(STATE_FILE)


def run(cmd, color="yellow", retries=2, arm_port=None, cam=None):
    """팔 명령 하나 실행 → 결과 dict. 진행 로그는 표준에러로."""
    from robot import devices as DEV
    from robot.arm import ArmStop

    LOCK_FILE.parent.mkdir(parents=True, exist_ok=True)
    lock = open(LOCK_FILE, "w")
    try:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError:
        return {"ok": False, "cmd": cmd, "result": "error", "error": "busy — 다른 팔 명령이 실행 중"}

    if cmd == "status":
        st = _load_state()
        return {"ok": True, "cmd": cmd, "result": "status", **st}

    from robot.tasks import ArmTasks
    tasks = None
    out = {"cmd": cmd}
    try:
        with contextlib.redirect_stdout(sys.stderr):
            tasks = ArmTasks(arm_port or DEV.ARM_PORT, cam or DEV.WRIST_CAM)
            st = _load_state()
            torque_on = tasks._torque_is_on()
            if torque_on:                               # 이전 실행이 블록을 든 채 끝남 → 이어받기
                tasks.holding = bool(st.get("holding"))
                tasks.pose = st.get("pose", "unknown")
            t0 = time.monotonic()
            if cmd == "pick":
                res = tasks.pick(color)
                for k in range(retries):
                    if res["result"] != "missed":
                        break
                    print(f"[다시 집기 {k+1}/{retries}] 못 쥠 → 다시 둘러보고 집기")
                    res = tasks.pick(color)
            elif cmd == "place":
                res = tasks.place()
            elif cmd == "park":
                res = tasks.park()
            else:
                raise ValueError(f"모르는 명령: {cmd}")
            out.update(res)
            out["seconds"] = round(time.monotonic() - t0, 1)
            out["ok"] = EXIT.get(res["result"], 2) == 0
    except ArmStop as e:
        out.update({"ok": False, "result": "error", "error": str(e)})
    except Exception as e:
        traceback.print_exc(file=sys.stderr)
        out.update({"ok": False, "result": "error", "error": f"{type(e).__name__}: {e}"})
    finally:
        if tasks is not None:
            with contextlib.redirect_stdout(sys.stderr):
                try:
                    out["holding"] = tasks.holding
                    out["pose"] = tasks.pose
                    _save_state(tasks)
                    if tasks.holding:
                        tasks.detach()                  # 블록을 든 채 → 토크 켠 채로 끝 (떨어뜨리지 않게)
                    elif out.get("result") == "error":
                        tasks.detach()                  # 오류 → 그 자리 유지 (사람이 확인)
                    else:
                        tasks.shutdown()                # 빈손 → 접힌 자세로 가서 토크 OFF
                        out["pose"] = "park"
                        tasks.pose = "park"
                        _save_state(tasks)
                except Exception as e:
                    print(f"[종료 처리 오류] {e}")
        lock.close()
    return out


def main():
    ap = argparse.ArgumentParser(description="팔 명령 실행 (끝나면 표준출력에 JSON 한 줄)")
    ap.add_argument("cmd", choices=["pick", "place", "park", "status"])
    ap.add_argument("--color", default="yellow")
    ap.add_argument("--retries", type=int, default=2)
    ap.add_argument("--arm-port")
    ap.add_argument("--cam")
    a = ap.parse_args()
    out = run(a.cmd, a.color, a.retries, a.arm_port, a.cam)
    print(json.dumps(out, ensure_ascii=False), flush=True)
    sys.exit(EXIT.get(out.get("result"), 0 if out.get("ok") else 2) if a.cmd != "status" else 0)


if __name__ == "__main__":
    main()

FILE robot/tasks.py
"""팔 작업 묶음 (팔 서버가 씀): 준비 → 찾아 집기 → 운반 자세 → 내려놓기 → 접힌 자세.

자세 4개 (관절값, calibration/*.json):
  park   접힌 자세 — 쉬거나 터틀봇이 빈손으로 이동할 때 (m5_teach_park.py)
  look   보는 자세 — 둘러보기 (m2)
  carry  운반 자세 — 블록을 든 채 터틀봇이 이동할 때. 라이다를 가리지 않게 (m6_teach_pose.py carry)
  place  내려놓는 자세 — 목적지에 도착한 뒤 이 자세에서 그리퍼를 열어 떨어뜨림 (m6_teach_pose.py place)
자세 사이 이동은 park.plan(관절 보간 + 상자·터틀봇 윗판 충돌 검사).

stop: 다른 스레드에서 request_stop() → 다음 단계 사이에서 멈춤(그 자리 유지). 한 동작(수백 ms~2초) 중간에는 못 끊음.
"""
import json
import threading
import time
from pathlib import Path

import numpy as np

from robot import mount as M
from robot import park as P
from robot import wristcam as WC
from robot.arm import ARM, Arm, ArmStop
from robot.kinematics import SO101Kinematics
from robot.motion import PASS_TOL, TRAVEL_TOL, go
from robot.pick import OPEN_PICK_PCT, PICK_CLEAR_M, PICK_RAISE_M
from robot.pickflow import go_down, plan_pick, unwind_roll
from vision.camera import UsbCamera

ROOT = Path(__file__).resolve().parents[1]
POSE_FILES = {"carry": ROOT / "calibration/carry_pose.json", "place": ROOT / "calibration/place_pose.json"}
LOGS = ROOT / "steps/logs"
CLOSED_PCT = 3
DROP_OPEN_PCT = 45         # 내려놓을 때 벌림 (떨어뜨리기)
HELD_MAX_PCT = 25          # 운반 중 그리퍼가 이보다 닫혀 있으면(블록 25mm ≈ 12%) 쥐고 있는 것


class Stopped(ArmStop):
    pass


def load_pose(name):
    f = POSE_FILES[name]
    if not f.is_file():
        return None
    q = json.loads(f.read_text(encoding="utf-8"))["q_deg"]
    return {j: float(q[j]) for j in ARM}


class ArmTasks:
    def __init__(self, port, cam):
        self.cal = WC.load_calib()
        self.look_q = WC.load_look_q()
        self.park_q = P.load_park()
        G = M.grasp_point()
        self.kin = SO101Kinematics()
        self.kin_g = SO101Kinematics(contact=G)
        self.kin_pick = SO101Kinematics(contact=G + np.array([-PICK_CLEAR_M, 0, 0]))
        print(f"[준비] 팔 연결 중 ({port})", flush=True)
        self.arm = Arm(port=port).connect()
        self.limits = {m: self.arm.range_deg(m) for m in ARM}
        self.cam_path = cam
        self.cam = None                 # 집을 때만 엶 (place·park는 카메라가 필요 없음 → 더 빨리 시작)
        print("[준비] 완료", flush=True)
        self.holding = False
        self.pose = "unknown"
        self._stop = threading.Event()
        self.stage_cb = lambda s: None

    # ------------------------------------------------------------ 공통
    def request_stop(self):
        self._stop.set()

    def _check(self, stage=None):
        if self._stop.is_set():
            self._stop.clear()
            try:
                self.arm.hold()
            except Exception:
                pass
            raise Stopped("중지 요청")
        if stage:
            self.stage_cb(stage)

    def _torque_is_on(self):
        return any(self.arm._read("Torque_Enable").values())

    def _goto_joint(self, q_to, name, last_tol=None):
        self._check(f"{name} 자세로")
        wps = P.plan(self.kin, self.arm.read_deg(), q_to)
        for i, q in enumerate(wps):
            self._check()
            last = i == len(wps) - 1
            res = self.arm.move_deg(q, correct=last, result_tol=last_tol if last else TRAVEL_TOL,
                                    pass_tol=None if last else PASS_TOL)
            if not res.status.startswith("도착"):
                raise ArmStop(res.status)
        self.pose = name


FILE steps/m5_teach_park.py
"""5단계(준비): 접힌 자세(park) 저장 — 한 번만. 이후 m4·미션이 이 자세에서 보는 자세로 스스로 펴고, 끝나면 여기로 접음.

  python steps/m5_teach_park.py

토크를 켜지 않고 읽기만 함. 팔을 평소 쉬는 '접힌 자세'(전원을 끄면 내려앉는 자세)에 둔 채 Enter.
저장 후 접힌 자세 → 보는 자세 경로를 미리 계산해 부딪힐 곳이 없는지 확인해 줌 (팔은 안 움직임).
"""
import json
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from robot import devices as DEV  # noqa: E402
from robot import park as P  # noqa: E402
from robot import wristcam as WC  # noqa: E402
from robot.arm import ARM, Arm, ArmStop  # noqa: E402
from robot.kinematics import SO101Kinematics  # noqa: E402


def main():
    arm = Arm(port=DEV.ARM_PORT).connect()
    try:
        if any(arm.torque_was.values()):
            print("[주의] 토크가 켜져 있음 — 다른 프로그램이 팔을 쓰는 중이면 끝내고 다시")
            return
        input("팔을 평소 쉬는 접힌 자세에 두고 Enter > ")
        q = arm.read_deg()
        print("  " + ", ".join(f"{j} {q[j]:+.1f}°" for j in ARM))
        out = {"q_deg": {j: q[j] for j in ARM}, "saved": time.strftime("%Y-%m-%d %H:%M:%S")}
        P.PARK_FILE.write_text(json.dumps(out, indent=2, ensure_ascii=False), encoding="utf-8")
        print(f"[저장] {P.PARK_FILE.relative_to(ROOT)}")
        tgt = P.park_target(arm, out["q_deg"])
        moved = {j: round(tgt[j] - q[j], 1) for j in ARM if abs(tgt[j] - q[j]) > 0.05}
        if moved:
            print(f"  측정 범위 끝에 붙은 관절은 목표를 안쪽으로 당겨 씀: {moved} (끝낼 때 토크 OFF하면 내려앉음)")
        try:
            kin = SO101Kinematics()
            wps = P.plan(kin, tgt, WC.load_look_q())
            print(f"  접힌 자세 → 보는 자세 경로 확인: 경유점 {len(wps)}개, 부딪힘 검사 통과")
        except ArmStop as e:
            print(f"  [경로] {e}")
    finally:
        arm.close(disable_torque=False)


if __name__ == "__main__":
    main()
